% ---------- Поля и межстрочный интервал (ГОСТ 2.105) ----------
% Поля по п. 2.9 методического пособия СЛИ (лист без рамки): левое 30 мм,

\usepackage[a4paper,left=30mm,right=10mm,top=20mm,bottom=20mm]{geometry}

\onehalfspacing
\usepackage{microtype}   % устраняет переполнение правого поля в justify-тексте
\emergencystretch=3em    % последний резерв: тянуть пробелы, а не вылезать за правое поле (актуально для \texttt-идентификаторов без переносов)

% ---------- Переносы слов (ГОСТ 2.105, п. 2.12: не рекомендуются) ----------
% Полный запрет не годится: без переноса TeX не может разбить длинное слово,
% и строка вылезает за правое поле (overfull hbox -- видимый дефект, хуже
% переноса). Штраф именно 10000 TeX трактует как "точка переноса запрещена
% вовсе"; 5000 -- всё ещё огромный штраф (перенос только там, где иначе
% строка не верстается), но конечный. \tolerance разрешает сильнее тянуть
% межсловные пробелы, лишь бы обойтись без переноса.
\hyphenpenalty=5000
\exhyphenpenalty=5000
\tolerance=2000

% В заголовках и ячейках таблиц перенос запрещён полностью (там строка всегда
% разбивается по словам, overfull не возникает).
\newcommand{\nohyphens}{\hyphenpenalty=10000\exhyphenpenalty=10000\relax}

% ---------- Абзацы ----------
% Абзацный отступ -- не менее 15 мм (пп. 2.6, 2.9 методического пособия СЛИ).
\setlength{\parindent}{1.5cm}
\setlength{\parskip}{0pt}
% Отдельная длина с тем же значением: titlesec обнуляет \parindent внутри
% заголовков, поэтому для отступа номера главы/подраздела нужна своя копия.
\newlength{\gostindent}
\setlength{\gostindent}{\parindent}

% ---------- Заголовки разделов ----------
% Каждый раздел верхнего уровня - с новой страницы; подразделы продолжают
% течь в потоке, без разрыва.
\usepackage{titlesec}
% \clearpage раньше стоял последним (пятым) аргументом \titleformat -
% "код перед заголовком" по документации titlesec. На практике в этой
% версии пакета \clearpage там выполняется МЕЖДУ номером раздела и его
% названием, а не перед всем заголовком: номер попадает на страницу ДО
% разрыва (осиротевшим внизу предыдущей страницы или один на пустой
% странице), а название - уже после разрыва, без номера. \pretocmd
% (etoolbox) вместо этого пришивает \clearpage перед самим \section
% целиком, до того как titlesec вообще начинает строить заголовок -
% разрыв происходит раньше, чем номер и название вообще собираются.
\usepackage{etoolbox}
\titleformat{\section}{\bfseries\raggedright\nohyphens}{\thesection}{0.5em}{}
\pretocmd{\section}{\clearpage}{}{}
\titleformat{\subsection}{\raggedright\nohyphens}{\thesubsection}{0.5em}{}
\titleformat{\subsubsection}{\raggedright\nohyphens}{\thesubsubsection}{0.5em}{}
\titlespacing*{\section}{0pt}{2\baselineskip}{1\baselineskip}
\titlespacing*{\subsection}{0pt}{1\baselineskip}{1\baselineskip}
\titlespacing*{\subsubsection}{0pt}{1\baselineskip}{0.5\baselineskip}
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

% ---------- Литеральное подчёркивание в идентификаторах (SENTRY_DSN и т.п.) ----------
\usepackage[strings]{underscore}

% ---------- Списки ----------
\usepackage{enumitem}
\setlist[itemize]  {noitemsep, topsep=0pt, parsep=0pt, leftmargin=0pt, itemindent=2.5cm, labelsep=1cm, label=--}
\setlist[enumerate]{itemsep=0pt, topsep=0pt, parsep=0pt, leftmargin=0pt, itemindent=2.7cm, labelsep=1cm, label=\arabic*)}

% ---------- Оглавление: точки-заполнители до номера страницы у разделов ----------
% По умолчанию в article.cls у \l@section (в отличие от \l@subsection) нет
% отточия -- копия стандартного определения с добавленным \leaders, как у
% \l@subsection.
\makeatletter
\renewcommand*\l@section[2]{%
  \ifnum \c@tocdepth >\z@
    \addpenalty\@secpenalty
    % 0.4em вместо стандартного 1.0em: с отбивкой в целый интервал
    % содержание не помещается на один лист.
    \addvspace{0.4em \@plus\p@}%
    \setlength\@tempdima{1.5em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode \bfseries
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      #1\nobreak
      \leaders\hbox{$\m@th\mkern \@dotsep mu.\mkern\@dotsep mu$}\hfill
      \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother

% ---------- Разделы с ручной нумерацией по ГОСТ (см. admin_manual/config.tex) ----------
% ---------- Заголовок приложения (пп. 12.3, 12.6 методического пособия) ----------
% Обозначение приложения -- по центру наверху листа, заголовок -- отдельной
% строкой симметрично тексту. Строка с характером приложения ("(справочное)",
% п. 12.5) по решению автора не печатается.
% #1 -- буква, #2 -- заголовок.
\newcommand{\gostappendix}[2]{%
  \clearpage
  \addcontentsline{toc}{section}{ПРИЛОЖЕНИЕ #1. #2}%
  \begin{center}
    {\bfseries ПРИЛОЖЕНИЕ #1}\\[\baselineskip]
    {\bfseries #2}
  \end{center}
  \par\vspace{0.5\baselineskip}
}
