% ---------- ГОСТ 2.105: разрыв таблицы между страницами ----------
% Требование: при переносе таблицы на следующую страницу нижнюю
% горизонтальную линию, ограничивающую таблицу, не проводят.
% longtable (основа xltabular) рисует каждый \hline как ДВЕ линейки подряд,
% разделённые \LT@sep = «штраф разрыва + \vskip-\arrayrulewidth». Если
% разрыва нет, отрицательный клей совмещает их и видна одна линия; если
% разрыв есть, первая линейка остаётся внизу страницы (та самая лишняя
% нижняя кромка), вторая уходит на начало следующей. Патч делает первую
% линейку нулевой высоты (без чернил) и убирает компенсирующий клей:
% суммарная высота и положение всех линий не меняются, но в месте переноса
% внизу страницы линии больше нет. Проверено попиксельно: позиции всех
% остальных линеек совпадают с непропатченным longtable.
\makeatletter
\ExplSyntaxOn
\cs_if_exist:NT \tbl_gdecr_row_count:
  {
    \def\LT@@hline{%
      \ifx\@let@token\hline
        \global\let\@gtempa\@gobble
        \gdef\LT@sep{\penalty-\@medpenalty\vskip\doublerulesep}%
      \else
        \global\let\@gtempa\@empty
        \gdef\LT@sep{\penalty-\@lowpenalty}%
      \fi
      \ifnum0=`{\fi}%
      \multispan\LT@cols\unskip\leaders\hrule\@height\z@\hfill\cr
      \noalign{\tbl_gdecr_row_count:\LT@sep}%
      \multispan\LT@cols\unskip\leaders\hrule\@height\arrayrulewidth\hfill\cr
      \noalign{\tbl_gdecr_row_count:\penalty\@M}%
      \@gtempa}%
  }
\ExplSyntaxOff
% Разрыв нельзя делать сразу ПОСЛЕ линейки \cline -- иначе она остаётся
% внизу страницы. Запрещаем разрыв в этом месте: TeX выбирает предыдущую
% допустимую точку (границу перед строкой-линейкой), и линейка уходит на
% следующую страницу вместе с продолжением таблицы. \nobreak обязан стоять
% ДО отрицательного клея, иначе разрыв всё равно возможен на самом клее,
% поэтому \patchcmd не годится -- определение \@cline повторено целиком
% (array.sty) с изменённой последней строкой.
\ExplSyntaxOn
\cs_if_exist:NT \tbl_gdecr_row_count:
  {
    \def\@cline#1-#2\@nil{
      \omit
      \@multicnt#1
      \advance\@multispan\m@ne
      \ifnum\@multicnt=\@ne\@firstofone{&\omit}\fi
      \@multicnt#2
      \advance\@multicnt-#1
      \advance\@multispan\@ne
      \UseTaggingSocket{tbl/leaders/begin}
      \leaders\hrule\@height\arrayrulewidth\hfill
      \UseTaggingSocket{tbl/leaders/end}
      \tbl_gdecr_row_count:
      \cr
      \noalign{\nobreak\vskip-\arrayrulewidth\nobreak}
    }
  }
\ExplSyntaxOff
\makeatother

% Три слоя вместо стандартных двух (background/main): лайнлайн участника
% рисуется на seqlifelayer -- позади всего, включая background, -- поэтому
% полоска активации (seqact, слой background, непрозрачная заливка) реально
% перекрывает и прячет дашед-лайнлайн под собой, а не просто "почти
% совпадает" с ним. Сам background по-прежнему нужен отдельно от main --
% иначе полоска активации рисуется поверх подписи условия alt, когда та
% выравнивается по x с лайнлайном участника.
\pgfdeclarelayer{seqlifelayer}
\pgfsetlayers{seqlifelayer,background,main}

% ---------- Диаграммы последовательностей (UML sequence, Приложение Б) ----------
% Минимальный набор макросов поверх tikz -- без отдельного пакета
% (tikz-uml), которого нет в дистрибутиве. x -- обычные числа (см),
% y отсчитывается автоматическим "курсором" \seqcur (длина), чтобы не
% считать вертикальные координаты вручную и не ловить наложения текста:
% каждая команда сама отступает от предыдущей строки на нужный отступ
% (с учётом того, растёт ли подпись стрелки вверх на одну или две строки).
\tikzset{
  seqactor/.style={draw, rectangle, minimum width=2.9cm, minimum height=0.9cm, text width=2.6cm, align=center, font=\small, fill=white},
  seqlife/.style={dashed, gray!60!black},
  seqact/.style={draw, fill=white},
  seqmsg/.style={-{Stealth}, font=\scriptsize, align=center},
  seqret/.style={-{Stealth}, dashed, font=\scriptsize, align=center},
}
\newlength{\seqcur}
% Начало диаграммы: курсор -- чуть ниже заголовков участников.
\newcommand{\seqinit}{\setlength{\seqcur}{-0.45cm}}
% Заголовок участника (верх, y=0).
% anchor=south/north вместо центрирования на (#1,0): при многострочной
% подписи (напр. "Клиентское\\приложение") бокс растёт от центра в обе
% стороны, и фиксированный отступ -0.45 попадал бы внутрь бокса, а линия
% жизни визуально пересекала бы его нижнюю границу.
\newcommand{\seqhead}[2]{\node[seqactor, anchor=south] at (#1,0) {#2};}
% Подвал участника на y=#3 (число, обычно из \seqmark) + линия жизни.
\newcommand{\seqfoot}[3]{%
  \node[seqactor, anchor=north] at (#1,#3) {#2};
  \begin{pgfonlayer}{seqlifelayer}\draw[seqlife] (#1,0) -- (#1,#3);\end{pgfonlayer}
}
% Отступ курсора вниз на #1 см (для ручных зазоров вокруг рамок).
\newcommand{\seqpad}[1]{\addtolength{\seqcur}{-#1cm}}
% Сохранить текущий курсор (в см, число) в макрос #1 -- для рамок alt/loop
% и полосок активации, которые ссылаются на уже пройденные строки.
\newcommand{\seqmark}[1]{\pgfmathsetmacro{#1}{\seqcur/1cm}}
% Синхронный вызов / возврат, подпись в одну строку. Оба конца стрелки
% сдвинуты на 0.09cm друг к другу -- ровно на половину ширины полоски
% активации (см. \seqact), так что стрелка упирается в боковую грань бокса
% участника, а не в сам лайнлайн, независимо от направления сообщения.
\newcommand{\seqmsgA}[3]{\seqpad{0.75}\seqmark{\seqy}\draw[seqmsg] ($(#1,\seqy)!0.09cm!(#2,\seqy)$) -- ($(#2,\seqy)!0.09cm!(#1,\seqy)$) node[midway, above] {#3};}
\newcommand{\seqretA}[3]{\seqpad{0.75}\seqmark{\seqy}\draw[seqret] ($(#1,\seqy)!0.09cm!(#2,\seqy)$) -- ($(#2,\seqy)!0.09cm!(#1,\seqy)$) node[midway, above] {#3};}
% То же, подпись в две строки (через \\).
\newcommand{\seqmsgB}[3]{\seqpad{1.1}\seqmark{\seqy}\draw[seqmsg] ($(#1,\seqy)!0.09cm!(#2,\seqy)$) -- ($(#2,\seqy)!0.09cm!(#1,\seqy)$) node[midway, above] {#3};}
\newcommand{\seqretB}[3]{\seqpad{1.1}\seqmark{\seqy}\draw[seqret] ($(#1,\seqy)!0.09cm!(#2,\seqy)$) -- ($(#2,\seqy)!0.09cm!(#1,\seqy)$) node[midway, above] {#3};}
% Условие alt/loop-ветки (короткая курсивная строка).
\newcommand{\seqcond}[2]{\seqpad{0.55}\seqmark{\seqy}\node[font=\scriptsize\itshape, align=center] at (#1,\seqy) {[#2]};}
% Самовызов (внутренняя обработка) справа от лайнлайна #1; #2 = подпись (1 строка).
\newcommand{\seqselfA}[2]{%
  \seqpad{0.3}\seqmark{\seqy}
  \draw[seqmsg] (#1+0.09,\seqy) -- ++(0.8,0) -- ++(0,-0.4) -- (#1+0.09,\seqy-0.4);
  \node[font=\scriptsize, align=left, anchor=north west, inner sep=0pt] at (#1+0.95,\seqy-0.4) {#2};
  \seqpad{0.75}
}
% То же, подпись в две строки.
\newcommand{\seqselfB}[2]{%
  \seqpad{0.3}\seqmark{\seqy}
  \draw[seqmsg] (#1+0.09,\seqy) -- ++(0.8,0) -- ++(0,-0.4) -- (#1+0.09,\seqy-0.4);
  \node[font=\scriptsize, align=left, anchor=north west, inner sep=0pt] at (#1+0.95,\seqy-0.4) {#2};
  \seqpad{1.1}
}
% Полоска активации между двумя отметками \seqmark (#2=верх, #3=низ, оба числа).
% На фоновом слое -- иначе полоска рисуется поверх подписи условия alt,
% когда та выравнивается по x с лайнлайном участника (see final_report).
\newcommand{\seqact}[3]{\begin{pgfonlayer}{background}\draw[seqact] (#1-0.09,#2) rectangle (#1+0.09,#3);\end{pgfonlayer}}
% Рамка фрагмента (alt/loop) между двумя отметками \seqmark: #1=подпись; #2,#3=левая/правая x; #4,#5=верх/низ.
\newcommand{\seqframe}[5]{%
  \draw (#2,#4) rectangle (#3,#5);
  \node[anchor=north west, font=\scriptsize\bfseries, inner sep=2pt] at (#2,#4) {#1};
}
\newcommand{\seqaltframe}[4]{\seqframe{alt}{#1}{#2}{#3}{#4}}
\newcommand{\seqloopframe}[4]{\seqframe{loop}{#1}{#2}{#3}{#4}}
\newcommand{\seqsep}[3]{\draw[seqlife] (#1,#2) -- (#3,#2);}

% ---------- Оглавление: точки-заполнители до номера страницы у разделов ----------
% По умолчанию в article.cls у \l@section (в отличие от \l@subsection) нет

\makeatletter
\renewcommand*\l@section[2]{%
  \ifnum \c@tocdepth >\z@
    \addpenalty\@secpenalty
    % 0.4em вместо стандартного 1.0em: с отбивкой в целый интервал
    % содержание не помещается на один лист.
    \addvspace{0.4em \@plus\p@}%
    \setlength\@tempdima{1.5em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode \bfseries
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      #1\nobreak
      \leaders\hbox{$\m@th\mkern \@dotsep mu.\mkern\@dotsep mu$}\hfill
      \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother
