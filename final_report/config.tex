% ---------- Заголовки разделов ----------
% Каждый раздел верхнего уровня - с новой страницы; подразделы продолжают
% течь в потоке, без разрыва.
\usepackage{titlesec}
% \clearpage раньше стоял последним (пятым) аргументом \titleformat -
% "код перед заголовком" по документации titlesec. На практике в этой
% версии пакета \clearpage там выполняется МЕЖДУ номером раздела и его
% названием, а не перед всем заголовком: номер попадает на страницу ДО
% разрыва (осиротевшим внизу предыдущей страницы или один на пустой
% странице), а название - уже после разрыва, без номера. \pretocmd
% (etoolbox) вместо этого пришивает \clearpage перед самим \section
% целиком, до того как titlesec вообще начинает строить заголовок -
% разрыв происходит раньше, чем номер и название вообще собираются.
\usepackage{etoolbox}
\titleformat{\section}{\bfseries\centering}{\thesection}{0.5em}{}
\pretocmd{\section}{\clearpage}{}{}
\titleformat{\subsection}{\bfseries\raggedright}{\thesubsection}{0.5em}{}
\titleformat{\subsubsection}{\bfseries\raggedright}{\thesubsubsection}{0.5em}{}
\titlespacing*{\section}{0pt}{2\baselineskip}{1\baselineskip}
\titlespacing*{\subsection}{0pt}{1.5\baselineskip}{0.75\baselineskip}
\titlespacing*{\subsubsection}{0pt}{1\baselineskip}{0.5\baselineskip}

% ---------- Диаграммы последовательностей (UML sequence) ----------
% Тот же макро-набор, что и в tech_zadanie/config.tex (Приложение Б) --
% курсор \seqcur сам отступает вниз на нужную величину после каждой
% команды, поэтому y-координаты не считаются вручную.
\tikzset{
  seqactor/.style={draw, rectangle, minimum width=2.9cm, minimum height=0.9cm, text width=2.6cm, align=center, font=\small, fill=white},
  seqlife/.style={dashed, gray!60!black},
  seqact/.style={draw, fill=white},
  seqmsg/.style={-{Stealth}, font=\scriptsize, align=center},
  seqret/.style={-{Stealth}, dashed, font=\scriptsize, align=center},
}
\newlength{\seqcur}
\newcommand{\seqinit}{\setlength{\seqcur}{-0.45cm}}
\newcommand{\seqhead}[2]{\node[seqactor] at (#1,0) {#2};}
\newcommand{\seqfoot}[3]{%
  \node[seqactor] at (#1,#3) {#2};
  \draw[seqlife] (#1,-0.45) -- (#1,#3+0.45);
}
\newcommand{\seqpad}[1]{\addtolength{\seqcur}{-#1cm}}
\newcommand{\seqmark}[1]{\pgfmathsetmacro{#1}{\seqcur/1cm}}
\newcommand{\seqmsgA}[3]{\seqpad{0.75}\seqmark{\seqy}\draw[seqmsg] (#1,\seqy) -- (#2,\seqy) node[midway, above] {#3};}
\newcommand{\seqretA}[3]{\seqpad{0.75}\seqmark{\seqy}\draw[seqret] (#1,\seqy) -- (#2,\seqy) node[midway, above] {#3};}
\newcommand{\seqmsgB}[3]{\seqpad{1.1}\seqmark{\seqy}\draw[seqmsg] (#1,\seqy) -- (#2,\seqy) node[midway, above] {#3};}
\newcommand{\seqretB}[3]{\seqpad{1.1}\seqmark{\seqy}\draw[seqret] (#1,\seqy) -- (#2,\seqy) node[midway, above] {#3};}
\newcommand{\seqcond}[2]{\seqpad{0.55}\seqmark{\seqy}\node[font=\scriptsize\itshape, align=center] at (#1,\seqy) {[#2]};}
\newcommand{\seqselfA}[2]{%
  \seqpad{0.3}\seqmark{\seqy}
  \draw[seqmsg] (#1+0.09,\seqy) -- ++(0.8,0) -- ++(0,-0.4) -- (#1+0.09,\seqy-0.4);
  \node[font=\scriptsize, align=left, anchor=north west, inner sep=0pt] at (#1+0.95,\seqy-0.4) {#2};
  \seqpad{0.75}
}
\newcommand{\seqselfB}[2]{%
  \seqpad{0.3}\seqmark{\seqy}
  \draw[seqmsg] (#1+0.09,\seqy) -- ++(0.8,0) -- ++(0,-0.4) -- (#1+0.09,\seqy-0.4);
  \node[font=\scriptsize, align=left, anchor=north west, inner sep=0pt] at (#1+0.95,\seqy-0.4) {#2};
  \seqpad{1.1}
}
% На фоновом слое -- иначе полоска рисуется поверх подписи условия alt,

\newcommand{\seqact}[3]{\begin{pgfonlayer}{background}\draw[seqact] (#1-0.09,#2) rectangle (#1+0.09,#3);\end{pgfonlayer}}
\newcommand{\seqframe}[5]{%
  \draw (#2,#4) rectangle (#3,#5);
  \node[anchor=north west, font=\scriptsize\bfseries, inner sep=2pt] at (#2,#4) {#1};
}

\newcommand{\seqloopframe}[4]{\seqframe{loop}{#1}{#2}{#3}{#4}}
\newcommand{\seqsep}[3]{\draw[seqlife] (#1,#2) -- (#3,#2);}
