% ---------- Узкие таблицы: подпись по левому краю самой таблицы ----------
% \caption (justification=raggedright выше) выравнивается по левому краю
% \linewidth, т.е. по полю страницы -- для таблицы уже textwidth это
% совпадает с левым краем таблицы, но для более узкой центрированной
% таблицы (тройка "Критерии и веса...", "Итоговые результаты...") подпись
% съезжает левее самой таблицы. \gosttablebox измеряет фактическую ширину
% таблицы (\lrbox), после чего подпись и таблица кладутся в minipage той
% же ширины -- у обеих общий левый край, а minipage целиком центрируется
% на странице тем же \centering, что раньше центрировал только таблицу.
\newsavebox{\gosttablebox}
\newenvironment{gosttable}[1]{%
  \def\gosttablecaption{#1}%
  \begin{lrbox}{\gosttablebox}%
}{%
  \end{lrbox}%
  \begin{minipage}{\wd\gosttablebox}
  \caption{\gosttablecaption}
  \usebox{\gosttablebox}
  \end{minipage}%
}

% ---------- Оглавление: точки-заполнители до номера страницы у разделов ----------
% По умолчанию в article.cls у \l@section (в отличие от \l@subsection) нет
% отточия -- копия стандартного определения с добавленным \leaders, как у
% \l@subsection.
\makeatletter
\renewcommand*\l@section[2]{%
  \ifnum \c@tocdepth >\z@
    \addpenalty\@secpenalty
    \addvspace{1.0em \@plus\p@}%
    \setlength\@tempdima{1.5em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode \bfseries
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      #1\nobreak
      \leaders\hbox{$\m@th\mkern \@dotsep mu.\mkern\@dotsep mu$}\hfill
      \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother
